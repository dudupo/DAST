\begin{definition} (connectivity)
  \label{def:connectivity}
\begin{enumerate}
  \item Let $G = (V, E)$ be an undirected graph. A \textbf{connected component} of $G$ is a subset $U \subseteq V$ in which there exists a path between every two vertices, and $U$ is 'maximal'\footnote{Local maximum.} in the sense that any extension of it by a vertex results in two vertices that are not connected.
\item A non-directed graph $G$ is said to be a \textbf{connected} graph if it only has one connected component.
\item Let $G = \left(V, E\right)$ be a directed graph. A \textbf{strongly connected component} of $G$ is a subset $U \subseteq V$ in which for any pair of vertices $u, v \in U$, there exist both a directed path from $u$ to $v$ and a directed path from $v$ to $u$, and $U$ is also 'maximal' in the sense that any extension of it by a vertex results in two vertices that are not connected (from both sides).
\item A directed graph $G$ is said to be a \textbf{strongly connected} if it only has one strongly connected component. 
\item A \textbf{directed acyclic graph (DAG)} is a directed graph with no cycles.
\end{enumerate}
\end{definition}
