\section{Shortest Paths} 

Recall that a weighted graph $G$ is a graph equipped with a weight function $w : E \rightarrow \mathbb{R}$, where the weight of a path is defined as the sum of the weights of the edges along the path. The shortest path between two vertices is a path that connects them and is also minimal in its weight. One native application of this formulation is to model routes between destinations, where edges correspond to roads and vertices to mid-stations.

Two main algorithms are used to solve the shortest path problem. The first is Dijkstra, which its correct only when the weight function is \textbf{non-negative} and runs in time $\Theta(|E| \log |V|)$ \footnote{Can be improved to $|E| + |V|\log|V|$, but that is beyond the scope of the course.}. The second is Bellman-Ford, which handles \textbf{negative} weights, yet it does so at a higher running time cost of $\Theta(|E| \cdot |V|)$.


Initially, estimators of the distances from the source are set to $\infty$, and during the process, their values are decreased until they reach the correct distance through the application of relaxation steps.



\begin{claim}
  \label{claim:subpath}
  Given a weighted (can be directed) graph $G$ with weights $w : E \rightarrow \mathbb{R}$, let $ v_{0},v_{1},..v_{k}$ be a shortest path from $v_{0}$ to $v_{k}$ then for any $0\le i<j \le k$ the path $v_{i},v_{i+1},..,v_{j-1},v_{j}$ is a shortest path from $v_{i}$ to $v_{j}$.
\end{claim}
\begin{proof}
  Assume by contradiction that there is a path $v_{i}, u_{i+1}, .., u_{j-1}, v_{j}$ that has a lower weight than $v_{i},v_{i+1},..,v_{j-1},v_{j}$ then: 
  
  \begin{equation*}
    \begin{split}
      w\left(     v_{0},v_{1},..v_{k} \right) &=w\left( v_{0},v_{1},..,v_{i}  \right) + w\left( v_{i},v_{i+1},..,v_{j} \right) + w\left( v_{j},v_{j+1},..,v_{k} \right) \\
      & > w\left( v_{0},v_{1},..,v_{i}  \right) + w\left( v_{i},u_{i+1},..,u_{j-1}, u_{j} \right) + w\left( v_{j},v_{j+1},..,v_{k} \right) \\
      & = w\left(     v_{0},v_{1},..,v_{i},u_{i+1},..,u_{j-1},v_{j}, .., v_{k} \right) 
    \end{split}
  \end{equation*}
\end{proof}

\textbf{Relaxation.} For $v, u \in V$, let $\delta(v, u)$ be the length of the shortest path from $v$ to $u$. Sometimes we will say that $\delta(v, u)$ is the distance between $v$ and $u$\footnote{Notice that for an undirected graph, the term distance is not well defined.}. For a source $s \in V$, let $d[v]$ be an estimator of $\delta(s, v)$. The relaxation method takes a directed edge $(u, v) \in E$ and updates $d[v]$ as in \Cref{alg:relax}. 

  \begin{algorithm}[h]
      \If{ $d[v] > d[u] + w(u,v)$ }{ 
        $d[v] \leftarrow d[u] + w(u,v)$ \\
        $\pi[v] \leftarrow u $ 
      }
    \caption{Relaxation.}
    \label{alg:relax}
  \end{algorithm} 

The standard algorithms to find shortest paths apply relaxation steps, where they differ in how they choose the next edge to be relaxed. Next, we show general properties that hold for all such algorithms, for both non-negative and possibly negative weights.

  \section{Shortest-Paths Properties} Let $G$ be (can be directed) a graph, For any $v,u \in E$ denote by $\delta(v,u)$ the length of a shortest path from $v$ to $u$. For a source vertex $s \in V$ and any vertex $v \in V $, associate an estimator $d[v]$ initialize as:  
  \begin{equation*}
    \begin{split}
      d[v] = \begin{cases}
        \infty & v \neq s \\
        0 & v = s
      \end{cases}
    \end{split}
  \end{equation*}
  Then the following properties hold:
  \begin{itemize}
    \item \textbf{Triangle inequality.}
        For all edges $(u,v) \in E$ we have: \begin{equation*}
          \begin{split}
            \delta(s,v) \le \delta(s,u) + w(u,v)
          \end{split}
        \end{equation*}
        \begin{proof}
          By definition, $\delta(s,v)$ is the weight of the minimal path from $s$ to $v$, Thus, it is lower or equal to the weight of any other path from $s$ to $v$, particularly the weight of the path obtained by chaining a shortest path from $s$ to $u$ with the edge from $u$ to $v$. The weight of such a path is $\delta(s,u) + w(u,v)$, and we get the desired inequality.
        \end{proof}

      \item \textbf{Upper-bound property.} 
        After the application of any sequence of relaxation steps, we have $d[v] \ge \delta(s,v)$.
          \begin{proof}
           By induction on the $i$th relaxation step, we present only the induction step. If $d[v]$ was changed after the step, then it was set to be $d[u] + w(u,v)$ for some $u \in V$ such that $(u,v) \in E$. However:
            \begin{equation*}
              \begin{split}
                d[u] + w(u,v) & \ge \delta(s,u) + w(u,v) \\ 
                & \ge \delta(s,v)
              \end{split}
            \end{equation*}
Where the first inequality uses the induction assumption for the $i-1$ previous relaxation steps, and the second follows from the triangle inequality.
          \end{proof}
      %\item \textbf{No-path property}
        %If there is  $\delta(s,v) = \infty$ 

        \item \textbf{Convergence property.} Suppose that a shortest path from $s$ to $v$ crosses trough $(u,v) \in E$ and suppose that $d[u] = \delta(s,u)$ then after application of relaxation $(u,v)$ we have $d[v] = \delta(s,v)$. 
          \begin{proof}
            By \cref{claim:subpath}, any sub path of a shortest path is also a shortest path between its two ends. Therefore $\delta(s,v) = \delta(s,u) + w(u,v)$, combining the given that $d[u]$ equals to $\delta(s,v)$ we get the desired equivalence. 
          \end{proof}

        \item \textbf{Path-relaxation property.}
        Let $s = v_{0},v_{1},v_{2},..,v_{k}$ be a shortest path from $s$ to $v_{k}$. After an application of a sequence relaxation steps that includes, in order, $(v_{0},v_{1}),(v_{1},v_{2}),..,(v_{k-1},v_{k})$ we have that $d[v_{k}] = \delta(s,v_{k})$.
      \begin{proof}
        By induction on the $i$th (relevant) relaxation step, namely: $(v_{i-1},v_{i})$. We present only the induction step. By the induction assumption after the $i-1$th relevant step, we have that  $d[v_{i-1}] = \delta(s,v_{i-1})$, In addition, the \textbf{Upper-bound property} guarantees that following relaxations steps can't decrease (change) that value. Therefore, at the beginning of the $i$th relevant relaxation step, $d[v_{i-1}]$ is still equals to $\delta(s,v_{i-1}$ and from the \textbf{Convergence property} the application of the $(v_{i-1},v_{i})$ relaxation sets $d[v_{i}]$ to be $\delta(s,v_{i})$.


      \end{proof}
  \end{itemize}
  \begin{remark}
    Corman defines a few more properties and uses them to prove that the returned predecessor forest $\pi$ can be used to restore the paths. In this recitation, we will be satisfied with proving that the algorithms compute the lengths of the shortest paths. Yet, the generalization is not hard.
  \end{remark}

\section{Dijkstra}
Dijkstra can be thought of as BFS where the FIFO queue is replaced by a min-heap. Each time we extract the closest unvisited vertex, we scan its edges and check for each of its neighbors if getting to the neighbor through the corresponding edge defines a shorter path than what is currently known. If so, we update the heap. \Cref{alg:dijkstra} gives an implementation.

\textbf{What's wrong with negative weights?} Consider the following graph, 


    \begin{figure}[h]
      \centering
      \input{tikz-negative-dij.tex}
      \caption{Example of a graph with negative weights, yet no negative cycles, that Dijkstra fails on. }
      \label{fig:negative}
    \end{figure}



  \begin{algorithm}[h]
    \For { $v \in V$ }{ 
      $d[v] \leftarrow \infty $  \\
      $\pi[v] \leftarrow $ null  \\
    }
    $d[s] \leftarrow 0 $\\
    $Q \leftarrow $ Heap($d[v_{1}], d[v_{2}], .. d[v_{n}]$)\\
    \While{ $Q$ is not empty }{
      $u \leftarrow $ extract-min $(Q)$ \\
      \For { $(u,v) \in E$ } {
        \If{ $d[v] > d[u] + w(u,v)$ }{ 
          $d[v] \leftarrow d[u] + w(u,v)$ \\
          Update $(Q, d[v])$\\
          $\pi[v] \leftarrow u $ 
        }
      }
    }
    \caption{Dijkstra.}
    \label{alg:dijkstra}
  \end{algorithm}

  \begin{claim}[Dijkstra Correctness]
At the time $u$ is extracted from $Q$, we have $d[u] = \delta(s,u)$.
  \end{claim}
  \begin{proof}
Assume for the sake of contradiction that the claim is false, and let $u$ be the first vertex such that at the time it is being extracted, the value of $d[u]$ is strictly greater than $\delta(s,u)$. If $u=s$, the contradiction is immediate. By the \textbf{Upper-bound property}, if $u$ is not reachable from $s$, then $d[u] > \infty$ for the whole running, and again the contradiction follows directly. Hence, it is left to deal with the case where $u$ is reachable from $s$. Let $x$ and $y$ be two adjacent vertices on a shortest path $s \leadsto u$, such that $x$ has been extracted from $Q$ before $u$ and $y$ not. Notice that $x$ can be equal to $s$ and $y$ can be equal to $u$.


From the assumption that $u$ is the first vertex to be extracted that violates the claim, we have $d[x]=\delta(s,x)$, and by the \textbf{Convergence property}, also $d[y] = \delta(s,y)$. Yet, by the correctness of min-heap extraction, we have $d[u] \le d[y]$. Combining that with $\delta(s,y) \le \delta(s,u)$ ($\star$), we have:
    
    \begin{equation*}
      \begin{split}
        & \delta(s,y) \le \delta(s,u) \le d[u] \le d[y] = \delta(s,y)\\ 
        \Rightarrow & \delta(s,u) = d[u]
      \end{split}
    \end{equation*}
In contradiction to $u$ being the first vertex to have a strictly greater $d[u]$ than $\delta(s,u)$ at the extraction time.
  \end{proof}

  \begin{exercise}
Where is the exact point at which the proof fails for negative weights?
  \end{exercise}
  \begin{solution}
At $(\star)$, the inequality $\delta(s,y) \le \delta(s,v)$ is correct only if we assume that $w \ge 0$. In that case, by the \textbf{Triangle inequality}, we have that for adjacent vertices on a shortest path $x, y$, we have that:
    \begin{equation*}
      \begin{split}
        \delta(s,x) = \delta(s,y) + w(y,x) \le \delta(s,y)
      \end{split}
    \end{equation*}
On the other hand, without the assumption, one can think about the graph that consists of a single negative edge.
  \end{solution}


  \section{Bellman-Ford}
  
Consider the graph with 4 vertices, denoted by \( V = \{1, 2, 3, 4\} \). Assume that the shortest path from 1 to 4 is \( 1, 3, 2, 4 \), and consider the application of the following relaxation steps:

  \begin{equation*}
    \begin{split}
      & (1,2),\inb{(1,3)},(1,4),(2,1),(2,3),(2,4),(3,1),(3,2),(3,4),(4,1),(4,2),(4,3)\\
      & (1,2),(1,3),(1,4),(2,1),(2,3),(2,4),(3,1),\inb{(3,2)},(3,4),(4,1),(4,2),(4,3)\\
      & (1,2),(1,3),(1,4),(2,1),(2,3),\inb{(2,4)},(3,1),(3,2),(3,4),(4,1),(4,2),(4,3)
      %& (1,2),(1,3),(1,4),(2,1),(2,3),(2,4),(3,1),(3,2),(3,4) \\
      %& (1,2),(1,3),(1,4),(2,1),(2,3),(2,4),(3,1),(3,2),(3,4)
    \end{split}
  \end{equation*}
  Each row contains relaxations for all edges. Thus, in terms of the \textbf{Path-relaxation property}, it ensures that by the end of the $i$th row, the $i$th relevant relaxation will be called. That idea gives an algorithm that computes the shortest paths for a graph with negative weights.


  \begin{algorithm}[h]
    \For { $v \in V$ }{ 
      $d[v] \leftarrow \infty $  \\
      $\pi[v] \leftarrow $ null  \\
    }
    $d[s] \leftarrow 0 $\\

    \For{ $i \in [n-1]$ }{
      \For { $(u,v) \in E$ } {
        \If{ $d[v] > d[u] + w(u,v)$ }{ 
          $d[v] \leftarrow d[u] + w(u,v)$ \\
          $\pi[v] \leftarrow u $ 
        }
      }
    }
    \For{ $(u,v) \in E$ }{
      \label{line:Belcon}
        \If{ $d[v] > d[u] + w(u,v)$ }{ 
          \Return Negative Cycle 
        }
    }
    \caption{Bellman–Ford.}
    \label{alg:bellman-ford}
  \end{algorithm}


  Showing that Bellman-ford detects a negative cycle is left as exercise.
