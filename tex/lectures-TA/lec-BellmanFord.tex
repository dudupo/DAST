\section{Dijkstra}

\begin{exercise}
  Give an example 
\end{exercise}


  \begin{algorithm}[h]
    \For { $v \in V$ }{ 
      $d[v] \leftarrow \infty $  \\
      $\pi[v] \leftarrow $ null  \\
    }
    $d[s] \leftarrow 0 $\\
    $Q \leftarrow $ Heap($d[v_{1}], d[v_{2}], .. d[v_{n}]$)\\
    \While{ $Q$ is not empty }{
      $u \leftarrow $ extract-min $(Q)$ \\
      \For { $(u,v) \in E$ } {
        \If{ $d[v] > d[u] + w(u,v)$ }{ 
          $d[v] \leftarrow d[u] + w(u,v)$ \\
          Update $(Q, d[v])$\\
          $\pi[v] \leftarrow u $ 
        }
      }
    }
    \caption{Dijkstra.}
    \label{alg:dijkstra}
  \end{algorithm}

  \begin{algorithm}[h]
    \For { $v \in V$ }{ 
      $d[v] \leftarrow \infty $  \\
      $\pi[v] \leftarrow $ null  \\
    }
    $d[s] \leftarrow 0 $\\

    \For{ $i \in [n]$ }{
      \For { $(u,v) \in E$ } {
        \If{ $d[v] > d[u] + w(u,v)$ }{ 
          $d[v] \leftarrow d[u] + w(u,v)$ \\
          $\pi[v] \leftarrow u $ 
        }
      }
    }
    \For{ $(u,v) \in E$ }{
      \label{line:Belcon}
        \If{ $d[v] > d[u] + w(u,v)$ }{ 
          \Return Negative Cycle 
        }
    }
    \caption{Bellman–Ford.}
    \label{alg:bellman-ford}
  \end{algorithm}


  \textbf{Detecting a Negative Cycle.} Suppose that $v_{1},v_{1},\cdots v_{k}=v_{1}$ is a negative cycle. This means that: 

  \begin{equation*}
    \begin{split}
      \sum^{k}_{i}w(v_{i},v_{i+1}) < 0 
    \end{split}
  \end{equation*}

  Assume that the algorithm doesn't detect the cycle, namely the condition at \Cref{line:Belcon} is not satisfied for any of the edges. Then: 

  \begin{equation*}
    \begin{split}
      & \forall v_{i} \ d[v] < d[u] + w(u,v) \\
      \Rightarrow & \sum^{k}_{i=2}d[v_{i}] < \sum^{k}_{i=2}d[v_{i-1}] + \sum^{k}_{i=2}w(v_{i-1},v_{i}) \\ 
      & =  \sum^{k}_{i=2}d[v_{i}] + \sum^{k}_{i=2}w(v_{i-1},v_{i}) 
    \end{split}
  \end{equation*}
